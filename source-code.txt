paths=$(pwd)

#!/bin/sh

# OS判定
OS=$(uname)

if [ "$OS" = "Linux" ]; then
echo 'export PATH="$PATH:$HOME/bin"' >> ~/.bashrc && source ~/.bashrc
fi

if [ "$OS" = "Darwin" ]; then
echo 'export PATH="$PATH:$HOME/bin"' >> ~/.zshrc && source ~/.zshrc

else

echo "このOSがbashを使っているか判定できないため確実なPATHを登録できない可能性があります。"
echo 'export PATH="$PATH:$HOME/bin"' >> ~/.bashrc && source ~/.bashrc
echo "処理は続行されました。"
fi

msgbox() {
    title="$1"
    message="$2"
    
    if [ "$(uname)" = "Darwin" ]; then
        # macOS: AppleScriptを使用して前面にダイアログを表示
        osascript -e "tell application \"System Events\" to display dialog \"$message\" with title \"$title\" buttons {\"OK\"} default button \"OK\"" >/dev/null 2>&1
    elif command -v zenity >/dev/null 2>&1; then
        # Linux (GUI): Zenityが使える場合
        zenity --info --title="$title" --text="$message" --width=300
    elif command -v whiptail >/dev/null 2>&1; then
        # Linux (CUI): ターミナル内でのメッセージボックス（Ubuntuなどに標準搭載）
        whiptail --title "$title" --msgbox "$message" 10 50
    elif command -v dialog >/dev/null 2>&1; then
        # Linux (CUI): dialogコマンドが使える場合
        dialog --title "$title" --msgbox "$message" 10 50
    else
        # 最終手段: どのコマンドもない場合はターミナルに出力
        echo "=== $title ==="
        echo "$message"
        echo "=============="
    fi
}
